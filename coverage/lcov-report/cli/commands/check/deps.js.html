/**
 * Check if a command is available
 * @param {string} command - Command to check
 * @returns {boolean} - True if available
 */
function checkCommand(command) {
  try {
    execSync(`command -v ${command}`, { stdio: 'ignore' });
    return true;
  } catch {
    return false;
  }
}
